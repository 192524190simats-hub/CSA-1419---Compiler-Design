int nc = 0;

int yywrap()
{
    return 1;
}

int main(int argc, char *argv[])
{
    if (argc < 2)
    {
        printf("Please provide a C file name.\n");
        return 1;
    }

    yyin = fopen(argv[1], "r");

    if (yyin == NULL)
    {
        printf("Cannot open input file.\n");
        return 1;
    }

    yyout = fopen("output.c", "w");

    yylex();

    printf("Number of comments = %d\n", nc);

    fclose(yyin);
    fclose(yyout);

    return 0;
}
